### 5. One small fix in `main.rs` + Backend Sweep PowerShell Script
Notice in `main.rs` above, `ui_event_subscription` needs to actually pull `UiEvent`s from `rx`! Swap `rx` in `NotchA` to a static `OnceLock<Mutex<Option<tokio::sync::mpsc::UnboundedReceiver<UiEvent>>>>` so `ui_event_subscription` forwards every `UiEvent` cleanly:
- Replace `rx: std::sync::Mutex<Option<...>>` + `ui_event_subscription` with a `static UI_RX: std::sync::OnceLock<std::sync::Mutex<Option<tokio::sync::mpsc::UnboundedReceiver<UiEvent>>>> = std::sync::OnceLock::new();`, store `rx` inside `boot()`, and in `ui_event_subscription`:
```rust
static UI_RX: std::sync::OnceLock<std::sync::Mutex<Option<tokio::sync::mpsc::UnboundedReceiver<UiEvent>>>> = std::sync::OnceLock::new();

fn ui_event_subscription(_app: &AppHandle) -> Subscription<Msg> {
    Subscription::run(|| {
        iced::stream::channel(256, |mut output: iced::futures::channel::mpsc::Sender<Msg>| async move {
            use iced::futures::SinkExt;
            let mut rx = UI_RX.get().and_then(|m| m.lock().unwrap().take());
            if let Some(ref mut receiver) = rx {
                while let Some(ev) = receiver.recv().await {
                    let _ = output.send(Msg::Ui(ev)).await;
                }
            }
            std::future::pending::<()>().await;
        })
    })
}
```
